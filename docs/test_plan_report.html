import pytest
import pandas as pd
import numpy as np

from core.risk_levels import calculate_risk_levels
from core.indicators import calculate_objective_indicators
from core.scoring import calculate_final_score, get_entry_signal


# ------------------------------------------------------------------
# Fixtures
# ------------------------------------------------------------------
@pytest.fixture
def mock_stock_data():
    """300일 치의 주가 및 거래량 샘플 데이터 생성"""
    np.random.seed(42)
    dates = pd.date_range(start="2025-01-01", periods=300, freq="D")
    
    close = pd.Series(np.linspace(10000, 15000, 300) + np.random.normal(0, 100, 300), index=dates)
    high = close + np.random.uniform(50, 200, 300)
    low = close - np.random.uniform(50, 200, 300)
    open_p = close + np.random.uniform(-50, 50, 300)
    volume = pd.Series(np.random.randint(100000, 1000000, 300), index=dates)

    foreign_data = {
        "net_values": [1000, -500, -2000],
        "consec_sell": 0,
        "consec_buy": 1,
        "recent_sum": -1500,
        "is_turning": True,
        "is_still_selling": False
    }

    news_data = {
        "today_count": 0,
        "today_ratio": 0.0
    }

    return {
        "close": close, "high": high, "low": low, "open": open_p,
        "volume": volume, "foreign": foreign_data, "news": news_data
    }


# ------------------------------------------------------------------
# core/risk_levels.py 테스트
# ------------------------------------------------------------------
class TestRiskLevels:
    def test_calculate_risk_levels_success(self, mock_stock_data):
        d = mock_stock_data
        curr_price = float(d["close"].iloc[-1])
        
        result = calculate_risk_levels(d["close"], d["high"], d["low"], curr_price)
        
        assert result is not None
        assert "stop_loss" in result
        assert "rr_ratio" in result
        assert result["stop_loss"] < curr_price
        assert result["target1"] > curr_price
        assert result["rr_verdict"] in ["우수", "양호", "미흡", "산출불가"]

    def test_calculate_risk_levels_exception_handling(self):
        # 데이터 부적합 시 None 리턴 검증
        empty_series = pd.Series([], dtype=float)
        result = calculate_risk_levels(empty_series, empty_series, empty_series, 10000)
        assert result is None


# ------------------------------------------------------------------
# core/indicators.py 테스트
# ------------------------------------------------------------------
class TestIndicators:
    def test_calculate_objective_indicators_structure(self, mock_stock_data):
        d = mock_stock_data
        results, score = calculate_objective_indicators(
            close_series=d["close"],
            volume_series=d["volume"],
            foreign_data=d["foreign"],
            news_data=d["news"],
            is_kosdaq=False,
            high_series=d["high"],
            low_series=d["low"],
            open_series=d["open"]
        )
        
        # 기본 반환 구조 및 0~80점 사이 오차범위 검증
        assert isinstance(results, dict)
        assert 0 <= score <= 80
        assert "rsi" in results
        assert "trend" in results
        assert "volume" in results

    def test_indicators_data_insufficient_fallback(self):
        # 부족한 스칼라/시리즈 전달 시 에러 방지 처리 검증
        short_series = pd.Series([100, 105, 102])
        results, score = calculate_objective_indicators(
            close_series=short_series,
            volume_series=short_series,
            foreign_data={},
            news_data={},
            is_kosdaq=True
        )
        assert results["rsi"]["status"] == "yellow"
        assert results["trend"]["is_structural_downtrend"] is False


# ------------------------------------------------------------------
# core/scoring.py 테스트
# ------------------------------------------------------------------
class TestScoring:
    def test_structural_downtrend_cap(self):
        # 200일선 구조적 하락추세 시 점수 상한선(55점) 적용 검증
        obj_indicators = {
            "rsi": {"value": 20},
            "w52": {"score": 15},
            "volume": {"score": 10},
            "pvd": {"score": 15},
            "bb": {"score": 15},
            "obv": {"score": 10},
            "foreign": {"score": 15},
            "drawdown": {"value": 35},
            "trend": {"is_structural_downtrend": True}
        }
        
        score, tier = calculate_final_score(obj_indicators, community_raw=50, is_kosdaq=False, fomo_score=15)
        assert score <= 55

    def test_get_entry_signal_falling_knife(self):
        # 구조적 하락 추세 + 공포 구간 조건에서 '진입 보류' 신호 반환 검증
        obj_indicators = {
            "foreign": {"is_turning": True, "is_still_selling": False},
            "obv": {"status": "green"},
            "pvd": {"score": 15},
            "rsi": {"value": 25},
            "trend": {"is_structural_downtrend": True}
        }
        
        signal = get_entry_signal(obj_indicators, final_scream_score=75, risk_levels=None)
        assert "진입 보류" in signal["level"]
        assert "구조적 하락추세" in signal["desc"]
